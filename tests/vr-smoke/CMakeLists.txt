cmake_minimum_required(VERSION 3.20)
project(Psvr2VrSmoke LANGUAGES CXX)
set(OPENVR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../external/CustomHeadsetOpenVR/ThirdParty/openvr")
if(NOT EXISTS "${OPENVR_ROOT}/headers/openvr.h")
  message(FATAL_ERROR "Initialize external/CustomHeadsetOpenVR ThirdParty/openvr submodule first.")
endif()
add_executable(psvr2_vr_smoke ../vr_smoke.cpp)
target_compile_features(psvr2_vr_smoke PRIVATE cxx_std_17)
target_compile_definitions(psvr2_vr_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(psvr2_vr_smoke PRIVATE "${OPENVR_ROOT}/headers")
target_link_libraries(psvr2_vr_smoke PRIVATE d3d11 dxgi d3dcompiler "${OPENVR_ROOT}/lib/win64/openvr_api.lib")
if(MSVC)
  target_compile_options(psvr2_vr_smoke PRIVATE /W4 /EHsc)
endif()
